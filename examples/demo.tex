\begin{frame}{Timed overlays: lightweight pseudo-animation}
  \transduration<1>{0.6}
  \transduration<2>{0.6}

  \centering
  \begin{tikzpicture}[x=2.8cm,y=1cm]
    \node[draw,rounded corners,minimum width=1.8cm,minimum height=0.9cm]
      (input) at (0,0) {Input};

    \node<2->[draw,rounded corners,minimum width=1.8cm,minimum height=0.9cm]
      (model) at (1,0) {Model};

    \node<3->[draw,rounded corners,minimum width=1.8cm,minimum height=0.9cm]
      (result) at (2,0) {Result};

    \draw<2->[->,thick] (input) -- (model);
    \draw<3->[->,thick] (model) -- (result);
  \end{tikzpicture}

  \vspace{1em}
  \only<1>{State 1: input}
  \only<2>{State 2: input $\rightarrow$ processing}
  \only<3>{State 3: input $\rightarrow$ processing $\rightarrow$ result --- auto-advance stops here}
\end{frame}
